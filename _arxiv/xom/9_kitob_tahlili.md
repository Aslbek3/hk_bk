## main.py

```python
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, CallbackQueryHandler, ContextTypes
import logging

from handlers.start import start
from handlers.buttons import button_handler
from handlers.files import send_file_handler

import config

logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO
)

async def main():
    app = ApplicationBuilder().token(config.TOKEN).build()

    # /start komandasi
    app.add_handler(CommandHandler("start", start))

    # Inline tugmalar uchun callback
    app.add_handler(CallbackQueryHandler(button_handler))

    # Fayl yuborish uchun komandalar yoki boshqa handlerlar
    app.add_handler(CommandHandler("sendfile", send_file_handler))

    print("🤖 Bot ishga tushdi!")
    await app.run_polling()

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```

## handlers/buttons.py

```python
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import ContextTypes, CallbackQueryHandler

# Inline tugmalar menyusi (misol uchun)
buttons = [
    [InlineKeyboardButton("Anime", callback_data="media_anime")],
    [InlineKeyboardButton("Kino", callback_data="media_kino")],
    [InlineKeyboardButton("Kitoblar", callback_data="media_books")]
]

async def button_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()

    if query.data == "media_anime":
        await query.edit_message_text(
            text="Anime bo‘limiga xush kelibsiz!",
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("Naruto", callback_data="file_naruto")],
                [InlineKeyboardButton("One Piece", callback_data="file_onepiece")]
            ])
        )
    elif query.data == "media_kino":
        await query.edit_message_text(
            text="Kino bo‘limi:",
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("Inception", callback_data="file_inception")],
                [InlineKeyboardButton("Interstellar", callback_data="file_interstellar")]
            ])
        )
    elif query.data == "media_books":
        await query.edit_message_text(
            text="Kitoblar bo‘limi:",
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("9 ta buyuk kitob", callback_data="file_9books")]
            ])
        )

    # Fayl tugmasi bosilganda fayl jo'natiladi
    elif query.data.startswith("file_"):
        file_key = query.data.split("_")[1]
        # Bu yerda fayl yuborish funksiyasini chaqirish kerak
        # Misol uchun: await send_file_by_key(query, file_key)
        await query.edit_message_text(text=f"Fayl yuborilmoqda: {file_key}")
```

```python
# Telegram API va handlerlar uchun
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup, ReplyKeyboardMarkup, KeyboardButton, InputFile
from telegram.ext import (
    ApplicationBuilder, CommandHandler, MessageHandler, filters,
    CallbackQueryHandler, ContextTypes, ConversationHandler
)
```

- **InlineKeyboardButton**, **InlineKeyboardMarkup** — inline tugmalar yaratish uchun.
- **ReplyKeyboardMarkup**, **KeyboardButton** — oddiy chatdagi tugmalar uchun.
- **InputFile** — fayl yuborishda ishlatiladi.
- **ApplicationBuilder** — botni ishga tushirish uchun.
- **CommandHandler** — `/start`, `/help` kabi komandalarni boshqarish uchun.
- **MessageHandler** — foydalanuvchi xabarlariga javob berish uchun.
- **filters** — xabarlarni filtrlash uchun (matn, fayl, rasm va hokazo).
- **CallbackQueryHandler** — inline tugmalar bosilganda ishga tushadi.
- **ContextTypes** — kontekstni boshqarish uchun.
- **ConversationHandler** — foydalanuvchi bilan dialog (ko‘p bosqichli suhbat) yaratish uchun.
